choropleth() の使用例

// choropleth() は分級→配色→レイヤー→凡例を1回の呼び出しで組み立てる
const { layer, legend } = Thematika.choropleth({
    data: geojson,
    value: 'POP_EST',            // プロパティ名または (f) => f.properties.POP_EST
    palette: 'YlOrRd',           // AllPalettes のパレット名
    classes: 5,                  // 階級数
    method: 'jenks',             // 'jenks' | 'equalInterval' | 'quantile' | 'stdDev'
    legend: {
        title: '人口（POP_EST）',
        position: { top: 20, left: 20 },
        labelFormat: d3.format('.2s')
    }
});

map.addLayer('choropleth', layer);
map.addLayer('legend', legend);

// タイトルとスケールバーで仕上げる
map.addLayer('title', new Thematika.TitleLayer({
    title: '世界の人口',
    subtitle: 'Natural Earth (POP_EST)',
    position: 'top-center'
}));
